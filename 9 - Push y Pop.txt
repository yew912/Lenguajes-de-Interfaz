; Descripcion: Practica sobre el uso de la pila mediante PUSH y POP para guardar datos y restaurarlos despues.

INCLUDE Irvine32.inc
.data
.code
main PROC
    ; metemos un 20 a eax y lo mandamos a la pila
    mov eax, 20

    push eax

    nop
    ; limpiamos ebx
    mov ebx, 0

    ; recuperamos de la pila hacia ebx
    pop ebx

    nop


    ; BLOQUE 2: LIFO

    mov eax, 10 ; ESP + 8
    push eax

    mov eax, 20 ; ESP + 4
    push eax

    mov eax, 30 ; ESP
    push eax

    nop
    ; al salir van cayendo al reves porque la pila saca ultimo lo primero que entra
    ; sacamos el 30 hacia eax
    pop eax
    ; sacamos el 20 hacia ebx
    pop ebx
    ; sacamos el 10 hacia ecx
    pop ecx

    ; BLOQUE 3: CONSERVAR REGISTROS
    ; guardamos valores originales para no perderlos
    mov eax, 100
    mov ebx, 200

    push eax
    push ebx

    mov eax, 999
    mov ebx, 666

    pop ebx
    pop eax

    ; BLOQUE 4: CALL Y RET

    mov eax, 111
    mov ebx, 222
    ; llamado a la subrutina
    call aumentarDiez



    nop

    exit
main ENDP

aumentarDiez PROC
    push ebx

    mov ebx, 10
    add eax, ebx

    pop ebx

    ret
aumentarDiez ENDP

END main
