.code
main PROC
; 1. ADD y SUB: Operaciones de suma y resta

mov eax, 50
mov ebx, 20

; le suma ebx a eax y luego le resta 15
add eax, ebx
sub eax, 15

nop

;2. CF: ACARREO EN UNA SUMA SIN SIGNO
mov eax, 0
mov al, 255

; al pasar el maximo de 8 bits prende el acarreo
add al,1

call dumpRegs

nop

;3. OF: DESCBORDAMIENTO EN SUMA CON SIGNO

mov eax, 0
mov al, 127
add al, 1

nop

;4. MUL: PRODUCTO DE VALORES DENTRO DE 32 BITS SIN SIGNO

mov eax, 25
mov ebx, 3

mul ebx

nop

; 5. IMUL: PRODUCTO DE VALORES CON SIGNO DENTRO DE 32 BITS

mov eax, -5
imul eax, eax,3

; 6. MUL: PRODUCTO DE VALORES QUE EXCEDEN 32 BITS

mov eax, 65537
mov ebx, 65537

mul ebx

nop

exit

main ENDP
END main
