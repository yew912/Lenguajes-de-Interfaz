; Descripción: Programa que usa el registro ECX como contador para repetir un ciclo con loop y mostrar una secuencia numérica.

INCLUDE Irvine32.inc
.data
.code
main PROC
    mov ecx, 5
ciclo:
    mov eax, ecx
    call WriteDec
    call Crlf
    loop ciclo
    exit
main ENDP
END main
