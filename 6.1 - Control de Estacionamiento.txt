; Descripcion: Control de un estacionamiento para revisar los lugares que quedan y validar que no entren mas de los que caben.

INCLUDE Irvine32.inc

.data
    msgEntrada BYTE "Ingrese cantidad de autos ",0
    msgOcupados BYTE "Espacios ocupados ",0
    msgDisponibles BYTE "Espacios disponibles ",0
    msgLleno BYTE "Estacionamiento lleno",0
    msgExceso BYTE "No hay suficientes espacios disponibles",0
    msgInvalido BYTE "Cantidad invalida",0

    ; capacidad maxima
    capacidad = 20
    ; variable para guardar los autos actualizados
    ocupados DWORD 0

.code

main PROC

ingresar:
    ; mensaje de bienvenida/pedir autos
    mov edx, OFFSET msgEntrada
    call WriteString
    ; lee numero de autos
    call ReadInt
    ; checar que el numero ingresado sea mayor a cero
    cmp eax, 0
    jle invalido

    ; pasamos el valor ingresado a ebx
    mov ebx, eax
    ; traemos los ocupados de memoria a eax
    mov eax, ocupados
    ; le sumamos la nueva cantidad
    add eax, ebx

    ; valida si pasa el cupo maximo
    cmp eax, capacidad
    jg exceso
    ; guarda la suma en la variable
    mov ocupados, eax
    
    ; muestra cuantos van ocupados
    mov edx, OFFSET msgOcupados
    call WriteString

    mov eax, ocupados
    call WriteDec
    call Crlf

    ; revisa si ya se lleno
    mov eax, ocupados
    cmp eax, capacidad
    je lleno

    ; despliega los que faltan
    mov edx, OFFSET msgDisponibles
    call WriteString

    ; restamos los ocupados de la capacidad
    mov eax, capacidad
    sub eax, ocupados
    call WriteDec
    call Crlf
    call Crlf

    jmp ingresar


exceso:

    ; mensaje de que sobrepasa el limite
    mov edx, OFFSET msgExceso
    call WriteString
    call Crlf
    call Crlf
    jmp ingresar


invalido:

    ; mensaje de error si ponen 0 o negativo
    mov edx, OFFSET msgInvalido
    call WriteString
    call Crlf
    call Crlf

    ; pide el dato otra vez
    jmp ingresar


lleno:

    ; mensaje de que ya se lleno
    mov edx, OFFSET msgLleno
    call WriteString
    call Crlf


fin:
    exit

main ENDP

END main
